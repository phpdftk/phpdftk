28/**

33 * no paint" rule.

495    /**
496     * Roots of `B'(t) = 0` for a cubic Bézier component, clipped to
497     * the open interval `(0, 1)` — endpoints are already covered by
498     * the segment's own endpoint contributions.
499     *
500     * @return list<float>
501     */
502    private static function cubicExtremaT(float $p0, float $p1, float $p2, float $p3): array
503    {
504        // B(t) = (1-t)³ p0 + 3(1-t)²t p1 + 3(1-t)t² p2 + t³ p3
505        // B'(t) = 3 [ (p1-p0) + 2t (p2 - 2p1 + p0) + t² (p3 - 3p2 + 3p1 - p0) ]
506        $a = $p3 - 3.0 * $p2 + 3.0 * $p1 - $p0;
507        $b = 2.0 * ($p2 - 2.0 * $p1 + $p0);
508        $c = $p1 - $p0;
509
510        if (abs($a) < self::EPSILON) {
511            // Reduced to linear: b·t + c = 0 → t = -c / b.
512            if (abs($b) < self::EPSILON) {
513                return [];
514            }
515            $t = -$c / $b;
516            return $t > 0.0 && $t < 1.0 ? [$t] : [];
517        }
518
519        $discriminant = $b * $b - 4.0 * $a * $c;
520        if ($discriminant < 0.0) {
521            return [];
522        }
523        $sqrtD = sqrt($discriminant);
524        $out = [];
525        foreach ([(-$b + $sqrtD) / (2.0 * $a), (-$b - $sqrtD) / (2.0 * $a)] as $t) {
526            if ($t > 0.0 && $t < 1.0) {
527                $out[] = $t;
528            }
529        }
530        return $out;
531    }
