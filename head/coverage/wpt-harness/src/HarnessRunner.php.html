323                status: TestStatus::HarnessError,
324                diffScore: 0.0,
325                reason: 'test file not found for ID',
326                diffArtefactPath: null,
327                renderMicros: 0.0,
328            );
329        }
330        $references = $this->references($testPath);
331        if ($references === []) {
332            return new TestResult(
333                testId: $testId,
334                status: TestStatus::Skipped,
335                diffScore: 0.0,
336                reason: 'no rel=match / rel=mismatch reference and no -ref.{png,html,xht,svg} sibling',
337                diffArtefactPath: null,
338                renderMicros: 0.0,
339            );
340        }
341
342        $start = hrtime(true);
343        try {
344            $renderedPng = $this->renderToPng($testPath);
345        } catch (\Throwable $e) {
346            return new TestResult(
347                testId: $testId,
348                status: TestStatus::Fail,
349                diffScore: 1.0,
350                reason: 'render failed: ' . $e->getMessage(),
351                diffArtefactPath: null,
352                renderMicros: (hrtime(true) - $start) / 1000.0,
353            );
354        }
355        $renderMicros = (hrtime(true) - $start) / 1000.0;
356
357        // "At least one match must match" and "all mismatches must
358        // mismatch" are two different quantifiers, so they are two
359        // separate walks.
360        $matches = array_values(array_filter(
361            $references,
362            static fn(ReftestReference $r) => $r->relation === ReferenceRelation::Match,
363        ));
364        $mismatches = array_values(array_filter(
365            $references,
366            static fn(ReftestReference $r) => $r->relation === ReferenceRelation::Mismatch,
367        ));
368
369        try {
370            $verdict = null;
371            foreach ($matches as $reference) {
372                $comparison = $this->compareAgainst($testPath, $renderedPng, $reference);
373                if ($comparison === null) {
374                    return $this->referenceRenderFailed($testId, $reference, $renderMicros, $verdict);
375                }
376                self::discard($verdict);
377                $verdict = $comparison;
378                if ($comparison['satisfied']) {
379                    break;
380                }
381            }
382            // No match reference held. Nothing downstream can rescue
383            // the test: a mismatch reference is an extra condition,
384            // never an alternative one.
385            if ($verdict !== null && !$verdict['satisfied']) {
386                return $this->fromVerdict($testId, $verdict, $renderMicros);
387            }
388
389            foreach ($mismatches as $reference) {
390                $comparison = $this->compareAgainst($testPath, $renderedPng, $reference);
391                if ($comparison === null) {
392                    return $this->referenceRenderFailed($testId, $reference, $renderMicros, $verdict);
393                }
394                if (!$comparison['satisfied']) {
395                    // Rendering the same as something the fixture says
396                    // it must differ from settles the whole test.
397                    self::discard($verdict);
398                    $verdict = $comparison;
399                    break;
400                }
401                // A satisfied mismatch only confirms a verdict the
402                // match walk already reached; keep it only when there
403                // was no match reference to reach one.
404                if ($verdict === null) {
405                    $verdict = $comparison;
406                } else {
407                    self::discard($comparison);
408                }
409            }
410
411            assert($verdict !== null);
412
413            return $this->fromVerdict($testId, $verdict, $renderMicros);
414        } finally {
415            @unlink($renderedPng);
416        }
417    }

419    /**
420     * Compare the already-rendered test frame against one reference.
421     * Returns null when the reference itself could not be rendered.
422     *
423     * @return array{diff: DiffResult, reference: ReftestReference, satisfied: bool}|null
424     */
425    private function compareAgainst(
426        string $testPath,
427        string $renderedPng,
428        ReftestReference $reference,
429    ): ?array {
430        try {
431            $refPng = str_ends_with(strtolower($reference->path), '.png')
432                ? $reference->path
433                : $this->renderToPng($reference->path);
434        } catch (\Throwable) {
435            return null;
436        }
437        $diff = $this->scorer->diff(
438            $renderedPng,
439            $refPng,
440            $this->fuzzyFor($testPath, $reference->path),
441        );
442        if ($refPng !== $reference->path) {
443            @unlink($refPng);
444        }
445
446        return [
447            'diff' => $diff,
448            'reference' => $reference,
449            // `$diff['passed']` is "the two frames are equal within
450            // whatever tolerance applies"; the relation decides what
451            // that means for the verdict.
452            'satisfied' => $reference->relation->satisfiedBy($diff['passed']),
453        ];
454    }

464
465        return new TestResult(

472            diffArtefactPath: $diff['diffImage'] ?? null,

476            // and can therefore be blank. That is not a blank pass,

479            bothRendersSolid: $verdict['satisfied'] && $diff['bothSolid'],

556    /**
557     * Resolve the `<meta name="fuzzy">` tolerance that applies when
558     * this test is compared against `$referencePath`.
559     *
560     * WPT selects fuzzy nodes exactly the way it selects reference
561     * links — `.//{http://www.w3.org/1999/xhtml}meta[@name='fuzzy']` —
562     * so a namespace prefix and unquoted attribute values are both
563     * normal, and 237 corpus fixtures do write `name=fuzzy` unquoted.
564     * A matcher that demands `name="fuzzy"` in quotes, and demands it
565     * before `content`, drops every one of those fixtures back onto
566     * the harness default threshold without saying so.
567     *
568     * A declaration may be keyed to one specific reference
569     * (`content="ref.html:0-5;0-100"` — WPT's `parse_ref_keyed_meta`,
570     * splitting on the last colon). A keyed declaration applies only
571     * to the reference it names; an unkeyed one is the default for
572     * every reference.
573     *
574     * @internal Exposed for {@see \Phpdftk\WptHarness\Tests\FuzzyMetaResolutionTest}.
575     */
576    public function fuzzyFor(string $testPath, string $referencePath): ?FuzzyTolerance
577    {
578        $head = @file_get_contents($testPath, false, null, 0, self::MARKUP_SCAN_BYTES);
579        if ($head === false || $head === '') {
580            return null;
581        }
582        $unkeyed = null;
583        foreach (self::elementsNamed($head, 'meta') as $tag) {
584            if (self::attributeValue($tag, 'name') !== 'fuzzy') {
585                continue;
586            }
587            $content = self::attributeValue($tag, 'content');
588            if ($content === null) {
589                continue;
590            }
591            $colon = strrpos($content, ':');
592            if ($colon !== false) {
593                $keyedRef = $this->resolveHref($testPath, substr($content, 0, $colon));
594                if ($keyedRef !== null) {
595                    if ($keyedRef === $referencePath) {
596                        $keyed = FuzzyTolerance::parse(substr($content, $colon + 1));
597                        if ($keyed !== null) {
598                            return $keyed;
599                        }
600                    }
601                    // Keyed at a different reference — not ours.
602                    continue;
603                }
604            }
605            $unkeyed ??= FuzzyTolerance::parse($content);
606        }
607        return $unkeyed;
608    }

621    /**
622     * The full reference list for a WPT reftest, in WPT's own order.
623     *
624     * WPT's manifest generator (`tools/manifest/sourcefile.py`)
625     * builds it from two `findall`s and one map:
626     *
627     *     match_links    = root.findall(".//{…xhtml}link[@rel='match']")
628     *     mismatch_links = root.findall(".//{…xhtml}link[@rel='mismatch']")
629     *     reftest_nodes  = match_links + mismatch_links
630     *     rel_map = {"match": "==", "mismatch": "!="}
631     *
632     * Four things follow, and this method honours all four:
633     *
634     *  1. The lookup is by **XHTML namespace**, not by literal tag
635     *     spelling. SVG and XML fixtures declare their reference as
636     *     `<html:link rel="match" href="…"/>` with `xmlns:html` bound
637     *     to the XHTML namespace — still an XHTML `link`, still a
638     *     reference. A matcher keyed on the literal string `<link`
639     *     misses every one of them (266 fixtures in the current
640     *     corpus, 112 of them under `css/css-masking/`), which drops
641     *     them out of scope entirely instead of scoring them.
642     *  2. The declared link is what makes a file a reftest, so when a
643     *     test declares one it **outranks** the legacy `<stem>-ref.*`
644     *     filename sibling. 35 fixtures in the corpus carry both and
645     *     disagree (e.g. `block-in-inline-remove-006.xht` declares
646     *     `…-nosplit-ref.xht` while a `…-ref.xht` sibling also
647     *     exists); WPT scores the declared one. `rel=mismatch` counts
648     *     for this too — it is every bit as much a reftest link as
649     *     `rel=match`, so a fixture declaring only mismatch
650     *     references is scored against those, not against a `-ref`
651     *     sibling it never pointed at (28 corpus fixtures).
652     *  3. `rel="mismatch"` is the negative relation: the test must
653     *     render *differently* from that reference. It is carried
654     *     here as {@see ReferenceRelation::Mismatch} rather than
655     *     dropped, which is what used to leave 375 real reftests
656     *     (286 css, 56 html, 4 svg, 29 mathml) unscored.
657     *  4. Every match link precedes every mismatch link regardless of
658     *     document order, because WPT concatenates the two `findall`
659     *     results in that order. Within a relation, document order is
660     *     the order the references get tried.
661     *
662     * The `<stem>-ref.*` sibling remains the fallback for a fixture
663     * that declares no reftest link at all: it is a CSS-WG-era
664     * convention that predates `rel=match`, and 53 corpus fixtures
665     * still rely on it alone. PNG wins among siblings since it
666     * short-circuits a re-render. There is deliberately no
667     * `<stem>-notref.*` counterpart. WPT classifies those files as
668     * references (`sourcefile.py::name_is_reference`, which is why
669     * they are skipped as tests) but never links a test to one by
670     * filename, so inventing that edge could only manufacture passes
671     * — and it would buy almost nothing anyway: 5 in-scope fixtures
672     * corpus-wide have a `-notref` sibling and no `rel=mismatch`.
673     *
674     * @return list<ReftestReference>
675     *
676     * @internal Exposed for {@see \Phpdftk\WptHarness\Tests\ReferenceResolutionTest}.
677     */
678    public function references(string $testPath): array
679    {
680        $declared = $this->declaredReftestReferences($testPath);
681        if ($declared !== []) {

683        }
684        $info = pathinfo($testPath);
685        $dir = $info['dirname'] ?? '.';
686        $stem = $info['filename'] ?? '';
687        $candidates = [
688            $dir . '/' . $stem . '-ref.png',
689            $dir . '/' . $stem . '-ref.html',
690            $dir . '/' . $stem . '-ref.xht',
691            $dir . '/' . $stem . '-ref.svg',
692        ];

694            $real = realpath($cand);
695            if ($real !== false && is_file($real)) {
696                return [new ReftestReference($real, ReferenceRelation::Match)];
697            }
698        }
699
700        return [];
701    }

725    /**
726     * Resolve every `<link rel="match">` / `<link rel="mismatch">` in
727     * a test file to an on-disk path, matches first.
728     *
729     * Tags are scanned in document order within each relation. A
730     * declaration whose href doesn't resolve to a file is dropped
731     * rather than allowed to poison the lookup — WPT has a served URL
732     * for every reference, we only have the checkout.
733     *
734     * Accepts any namespace prefix on the element name (`html:link`,
735     * `x:link`, …) and unquoted attribute values (`rel=match`), both
736     * of which occur in the corpus.
737     *
738     * Read is bounded to {@see self::MARKUP_SCAN_BYTES} so a
739     * malformed test can't stall the harness. Verified against the
740     * full corpus: zero fixtures declare their reference beyond that
741     * offset.
742     *
743     * @return list<ReftestReference>
744     */
745    private function declaredReftestReferences(string $testPath): array
746    {
747        $head = @file_get_contents($testPath, false, null, 0, self::MARKUP_SCAN_BYTES);
748        if ($head === false || $head === '') {
749            return [];
750        }
751        $tags = self::elementsNamed($head, 'link');
752        $references = [];
753        foreach ([ReferenceRelation::Match, ReferenceRelation::Mismatch] as $relation) {
754            $rel = $relation === ReferenceRelation::Match ? 'match' : 'mismatch';
755            foreach ($tags as $tag) {
756                if (self::attributeValue($tag, 'rel') !== $rel) {
757                    continue;
758                }
759                $href = self::attributeValue($tag, 'href');
760                if ($href === null) {
761                    continue;
